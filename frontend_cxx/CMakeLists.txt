cmake_minimum_required(VERSION 3.20)
project(frontend_cxx CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O2 -g")
set(TP ${CMAKE_SOURCE_DIR}/third_party/install)
include_directories(${CMAKE_SOURCE_DIR}/src ${TP}/include)
link_directories(${TP}/lib ${TP}/lib64)
enable_testing()

# One subdirectory per component; each owner keeps its own src/<c>/CMakeLists.txt and
# tests/<c>/CMakeLists.txt (see DESIGN.md).
set(FE_COMPONENTS common tokenizer render parser image engine http api)
set(FE_ADDED_SUBDIRS)
foreach(c ${FE_COMPONENTS})
  if(EXISTS ${CMAKE_SOURCE_DIR}/src/${c}/CMakeLists.txt)
    add_subdirectory(src/${c})
    get_property(_subs DIRECTORY ${CMAKE_SOURCE_DIR}/src/${c} PROPERTY SUBDIRECTORIES)
    list(APPEND FE_ADDED_SUBDIRS ${_subs})
  endif()
endforeach()
# tests/<c> is added here unless the component's own CMakeLists already added it.
foreach(c ${FE_COMPONENTS} e2e)
  if(EXISTS ${CMAKE_SOURCE_DIR}/tests/${c}/CMakeLists.txt AND
     NOT "${CMAKE_SOURCE_DIR}/tests/${c}" IN_LIST FE_ADDED_SUBDIRS)
    add_subdirectory(tests/${c})
  endif()
endforeach()

if(EXISTS ${CMAKE_SOURCE_DIR}/src/main.cpp)
  add_executable(frontend_cxx src/main.cpp)
  set(FE_LINK)
  foreach(lib fe_api fe_http fe_engine fe_parser fe_render fe_image fe_tokenizer fe_common)
    if(TARGET ${lib})
      list(APPEND FE_LINK ${lib})
    endif()
  endforeach()
  target_link_libraries(frontend_cxx PRIVATE ${FE_LINK} pthread)
endif()
